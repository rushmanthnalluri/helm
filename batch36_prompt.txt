### 5. Supervisor-First Foreground Process & UIA Browser Domain Resolution

- `current_proc` is resolved strictly via `windows_tools.get_foreground_window()`. The supervisor **never** falls back to worker-attested `args.get("target_process")`.
- When the foreground process is a browser, `current_domain` is resolved via Windows UI Automation (`uiautomation`) directly querying the browser address bar (Omnibox). Worker-attested `args.get("domain")` and window titles are strictly ignored.
```python
def get_browser_active_domain(hwnd: Optional[int] = None) -> Optional[str]:
    """Inspects active browser window via Windows UI Automation (UIA) to extract true URL hostname.
    Guarantees: Never trusts window titles or worker-supplied domain arguments.
    """
    try:
        import uiautomation as uia
        import urllib.parse
        
        edit = None
        if hwnd:
            try:
                window_ctrl = uia.ControlFromHandle(hwnd)
                if window_ctrl:
                    edit = window_ctrl.EditControl(searchDepth=8, SubName="Address")
                    if not edit.Exists(0.1):
                        edit = window_ctrl.EditControl(searchDepth=8, ClassName="Chrome_OmniboxView")
            except Exception:
                pass

        if not edit or not edit.Exists(0.1):
            edit = uia.EditControl(searchDepth=6, SubName="Address")
            if not edit.Exists(0.1):
                edit = uia.EditControl(searchDepth=6, ClassName="Chrome_OmniboxView")

        if edit and edit.Exists(0.1):
            val = edit.GetValuePattern().Value if hasattr(edit, 'GetValuePattern') else ""
            if val:
                val = str(val).strip()
                if not val.startswith(("http://", "https://")):
                    val = "https://" + val
                parsed = urllib.parse.urlparse(val)
                return parsed.hostname
    except Exception as e:
        logger.debug("UIA browser domain inspection error: %s", e)
    return None
```

        """Low-level keyboard hook callback.
        Guarantees physical console presence for pairing hotkey (Ctrl + Alt + P).
        Rejects all synthetic/injected keystrokes (LLKHF_INJECTED | LLKHF_LOWER_IL_INJECTED).
        Dispatches pairing asynchronously to worker thread to prevent LowLevelHooksTimeout.

        import ctypes
        from ctypes import wintypes
        user32 = ctypes.windll.user32
        WM_KEYDOWN = 0x0100
        WM_KEYUP = 0x0101
        WM_SYSKEYDOWN = 0x0104
        WM_SYSKEYUP = 0x0105
        LLKHF_INJECTED = 0x00000001
        LLKHF_LOWER_IL_INJECTED = 0x00000002

        if nCode >= 0:
            class KBDLLHOOKSTRUCT(ctypes.Structure):
                _fields_ = [
                    ("vkCode", wintypes.DWORD),
                    ("scanCode", wintypes.DWORD),
                    ("flags", wintypes.DWORD),
                    ("time", wintypes.DWORD),
                    ("dwExtraInfo", ctypes.c_ulong),
                ]
            kb = KBDLLHOOKSTRUCT.from_address(lParam)
            is_injected = bool(kb.flags & (LLKHF_INJECTED | LLKHF_LOWER_IL_INJECTED))
            if is_injected:
                # Reject any synthesized input attempting to trigger pairing ceremony
                if self._ll_hook:
                    return user32.CallNextHookEx(self._ll_hook, nCode, wParam, lParam)
                return 0

            vk = kb.vkCode
            if wParam in (WM_KEYDOWN, WM_SYSKEYDOWN):
